! ===== ISP (Cisco 2911) - Lab 11 baseline =====
enable
configure terminal
hostname ISP
no ip domain-lookup
interface gig0/1
 ip address 203.0.113.1 255.255.255.252
 no shutdown
interface gig0/0
 ip address 198.51.100.1 255.255.255.0
 no shutdown
end
write memory

! ===== End devices (Desktop > IP Configuration) =====
! PC1 (Fa0/1, VLAN 10 SALES) : DHCP
! PC2 (Fa0/2, VLAN 10 SALES) : DHCP
! PC3 (Fa0/3, VLAN 20 HR)    : DHCP
!
! LogServer (Fa0/4, VLAN 30) : static
!   IP 192.168.30.10  Mask 255.255.255.0  Gateway 192.168.30.1
!   Services > SYSLOG = On
!
! Internet server (behind ISP): static
!   IP 198.51.100.10  Mask 255.255.255.0  Gateway 198.51.100.1

! ===== Cabling =====
! PC1 - SW1 Fa0/1        straight-through
! PC2 - SW1 Fa0/2        straight-through
! PC3 - SW1 Fa0/3        straight-through
! LogServer - SW1 Fa0/4  straight-through
! SW1 Gi0/1 - R1 Gi0/0   straight-through
! R1 Gi0/1 - ISP Gi0/1   crossover
! ISP Gi0/0 - Internet   crossover
